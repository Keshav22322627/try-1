function AreaWiseData({ areaData, loading, expandedAreas, toggleArea, getFilteredUsers, getRoleBadge, getStatusBadge }) {
  if (loading) {
    return (
      <div className="flex justify-center py-12">
        <div className="animate-spin rounded-full h-8 w-8 border-b-2 border-[#002C6C]" />
      </div>
    );
  }

  if (areaData.length === 0) {
    return (
      <div className="text-center py-12">
        <Users className="w-12 h-12 text-slate-300 mx-auto mb-4" />
        <p className="text-slate-500">No staff or dealers found for your access level.</p>
      </div>
    );
  }

  return (
    <div className="space-y-4">
      {areaData.map((area) => {
        const isExpanded = expandedAreas[area.areaId];
        const filteredUsers = getFilteredUsers(area);
        
        return (
          <div key={area.areaId} className="bg-white border border-slate-200 rounded-2xl overflow-hidden">
            <button
              onClick={() => toggleArea(area.areaId)}
              className="w-full p-4 flex items-center justify-between hover:bg-slate-50 transition"
            >
              <div className="flex items-center gap-3">
                <MapPin className="w-5 h-5 text-[#002C6C]" />
                <div>
                  <h3 className="font-bold text-slate-900 text-lg">{area.areaName}</h3>
                  <p className="text-xs text-slate-500">
                    {area.totalStaff} Staff • {area.totalDealers} Dealers
                  </p>
                </div>
              </div>
              <div className="flex items-center gap-4 text-sm text-slate-500">
                <span className="flex items-center gap-1">
                  <Shield className="w-3.5 h-3.5 text-blue-500" />
                  {area.salesHeads.length} Heads
                </span>
                <span className="flex items-center gap-1">
                  <UserCheck className="w-3.5 h-3.5 text-green-500" />
                  {area.salesPersons.length} Persons
                </span>
                <span className="flex items-center gap-1">
                  <Building2 className="w-3.5 h-3.5 text-purple-500" />
                  {area.dealers.length} Dealers
                </span>
                {isExpanded ? (
                  <ChevronUp className="w-5 h-5 text-slate-400" />
                ) : (
                  <ChevronDown className="w-5 h-5 text-slate-400" />
                )}
              </div>
            </button>

            {isExpanded && filteredUsers.length > 0 && (
              <div className="border-t border-slate-100 p-4">
                <div className="overflow-x-auto">
                  <table className="w-full text-left text-sm border-collapse">
                    <thead className="bg-slate-50 text-slate-600 font-semibold uppercase border-b border-slate-200">
                      <tr>
                        <th className="p-3">Avatar</th>
                        <th className="p-3">Name</th>
                        <th className="p-3">Role</th>
                        <th className="p-3">Contact</th>
                        <th className="p-3">Business / GSTIN</th>
                        <th className="p-3">Status</th>
                      </tr>
                    </thead>
                    <tbody className="divide-y divide-slate-100">
                      {filteredUsers.map((user) => (
                        <tr key={user.id} className="hover:bg-slate-50">
                          <td className="p-3">
                            <img
                              src={user.avatar}
                              alt={user.name}
                              className="w-10 h-10 rounded-full object-cover border border-slate-200"
                              onError={(e) => { e.target.src = 'https://images.unsplash.com/photo-1534528741775-53994a69daeb?auto=format&fit=crop&q=80&w=250'; }}
                            />
                          </td>
                          <td className="p-3 font-bold text-slate-900">{user.name}</td>
                          <td className="p-3">{getRoleBadge(user.role)}</td>
                          <td className="p-3">
                            <div className="space-y-1 text-xs text-slate-600">
                              <div className="flex items-center gap-1">
                                <Mail className="w-3 h-3" />
                                {user.email}
                              </div>
                              <div className="flex items-center gap-1">
                                <Phone className="w-3 h-3" />
                                {user.phone}
                              </div>
                            </div>
                          </td>
                          <td className="p-3 text-xs text-slate-600">
                            {user.businessName && (
                              <div className="font-medium">{user.businessName}</div>
                            )}
                            {user.gstin && (
                              <div className="font-mono text-slate-500">{user.gstin}</div>
                            )}
                            {!user.businessName && !user.gstin && (
                              <span className="text-slate-400">—</span>
                            )}
                          </td>
                          <td className="p-3">{getStatusBadge(user.status)}</td>
                        </tr>
                      ))}
                    </tbody>
                  </table>
                </div>
              </div>
            )}
            
            {isExpanded && filteredUsers.length === 0 && area.totalStaff + area.totalDealers > 0 && (
              <div className="border-t border-slate-100 p-4 text-center text-slate-500">
                No users match the current filters in this area.
              </div>
            )}
          </div>
        );
      })}
    </div>
  );
}
